# Funzione per calcolare il GVF
calc_gvf <- function(x, n) {
  # Calcola i tagli con Jenks
  intervals <- classIntervals(x, n = n, style = "jenks")
  # Assegna ogni valore al suo gruppo
  groups <- cut(x, breaks = intervals$brks, include.lowest = TRUE)

  # Calcola Somma dei Quadrati Totale (SST)
  sst <- sum((x - mean(x))^2)
  # Calcola Somma dei Quadrati entro i gruppi (SSW)
  ssw <- sum(tapply(x, groups, function(val) sum((val - mean(val))^2)))

  # Formula GVF: (SST - SSW) / SST
  gvf <- (sst - ssw) / sst
  return(gvf)
}
