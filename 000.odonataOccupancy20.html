Sampling effort

Per tenere conto anche del differente sforzo di copertura delle celle del territorio nel corso degli anni, il modello lm è stato modificato in glm con una correzione derivante dall’utilizzo di un offset pari al log(tot_occurrences) per ciascun anno. Il modello normalizza i conteggi per lo sforzo, così che la pendenza stimata per date_year rifletta la vera tendenza di espansione o contrazione e non solo la variazione derivante dal differente sforzo di campionamento.

```{r glm_offset}
all_cells <- sort(unique(dfo$cellcode20))
all_years <- sort(unique(dfo$date_year))

effort_df <- dfo %>%
  count(cellcode20, date_year, name = 'tot_occurrences') %>%
  complete(
    cellcode20 = all_cells,
    date_year = all_years,
    fill = list(tot_occurrences = 0)

glmSpecies <- function(nomeSp) {
  y_df <- dfo %>%
    filter(species == nomeSp) %>%
    count(cellcode20, date_year, name = 'n_occurrences') %>%
    complete(
      # includi anche le celle mai occupate dalla specie
      cellcode20 = all_cells,
      date_year = all_years,
      fill = list(n_occurrences = 0)
    ) %>%
    mutate(detection = ifelse(n_occurrences > 0, 1, 0))

  species_sites <- merge(y_df, effort_df)

  model_offset <- glm(
    detection ~ date_year + offset(log(tot_occurrences + 1)),
    family = binomial,
    data = species_sites
  )

  sp_results <- tidy(model_offset)
  sp_results$species = nomeSp
  return(sp_results)
}

results <- map(listaSpecie, glmSpecies) |> list_rbind()

species_trends <- results |>
  filter(term == "date_year") |>
  mutate(
    trend = case_when(
      estimate > 0 ~ "increase",
      estimate < 0 ~ "decrease",
      TRUE ~ "stable"
    )
  )
```

Modelli di occupancy con unmarked

Modello colext

Il modello utilizzato è del tipo Dynamic occupancy models colext. La formula impiegata è colext(~1, ~year, ~year, ~effort, umf e considera una p in funzione dell’effort e una ψ (probabilità di occupazione), gamma (probabilità di colonizzazione - in blu), epsilon (probabilità di estinzione in rosso) costanti.

Code 
```{r unmarked}
#| eval: true

library(unmarked)

listaSpecie <- dfo |>
  count(species) |>
  filter(n >= 50) |>
  pull(species)

nomeSp <- listaSpecie[45]

occProb <- function(nomeSp) {
  #print(nomeSp)

  # 1. PREPARAZIONE DATI
  all_cells <- sort(unique(dfo$cellcode20))
  all_years <- sort(unique(dfo$date_year))

  # Matrice di rilevamento (y)
  y_matrix <- dfo %>%
    filter(species == nomeSp) %>%
    count(cellcode20, date_year, name = 'n_occurrences') %>%
    complete(
      # includi anche le celle mai occupate dalla specie
      cellcode20 = all_cells,
      date_year = all_years,
      fill = list(n_occurrences = 0)
    ) %>%
    mutate(detection = ifelse(n_occurrences > 0, 1, 0)) %>%
    pivot_wider(
      id_cols = cellcode20,
      names_from = date_year,
      values_from = detection
    ) %>%
    column_to_rownames("cellcode20") %>%
    as.matrix()

  # Matrice dello sforzo
  effort_matrix <- dfo %>%
    count(cellcode20, date_year, name = 'tot_occurrences') %>%
    complete(
      cellcode20 = all_cells,
      date_year = all_years,
      fill = list(tot_occurrences = 0)
    ) %>%
    pivot_wider(
      id_cols = cellcode20,
      names_from = date_year,
      values_from = tot_occurrences
    ) %>%
    column_to_rownames("cellcode20") %>%
    as.matrix()

  # Standardizzazione vettoriale
  effort_vec <- as.numeric(effort_matrix)
  effort_std_mat <- matrix(
    (effort_vec - mean(effort_vec)) / sd(effort_vec),
    nrow = nrow(y_matrix),
    ncol = ncol(y_matrix)
  )

  # Covariata Anno
  anni_num <- as.numeric(colnames(y_matrix))
  year_std <- as.vector(scale(anni_num))
  n_siti <- nrow(y_matrix)
  n_anni <- ncol(y_matrix)

  # matrice degli anni (standardizzati) con dimensioni siti x anni
  year_mat <- matrix(rep(year_std, each = n_siti), nrow = n_siti, ncol = n_anni)

  # 2. creazione umf
  umf <- unmarkedMultFrame(
    y = y_matrix,
    yearlySiteCovs = list(
      effort = effort_std_mat,
      year_idx = year_mat # Ora il modello troverà 'year_idx'
    ),
    numPrimary = n_anni
  )

  # 3. Esecuzione del modello
  m1 <- colext(
    psiformula = ~1,
    gammaformula = ~year_idx, # Tasso di colonizzazione influenzato dall'anno
    epsilonformula = ~year_idx, # Tasso di estinzione influenzato dall'anno
    pformula = ~effort,
    data = umf
  )

  # summary(m1)
  # Lo stato 1 è "non occupato", lo stato 2 è "occupato"
  sm <- smoothed(m1)
  occ_values <- as.numeric(sm["occupied", ])

  df_plot <- data.frame(
    Species = nomeSp,
    Anno = anni_num,
    Occupancy = occ_values
  )

  pOccu <- df_plot |>
    ggplot(aes(x = Anno, y = Occupancy)) +
    geom_line(color = "steelblue", linewidth = 1) +
    geom_point(color = "steelblue", size = 3) +
    geom_smooth(
      method = "loess",
      color = "red",
      linetype = "dashed",
      se = FALSE,
      alpha = 0.5
    ) +
    scale_y_continuous(limits = c(0, 1), labels = scales::percent) +
    labs(
      title = paste(nomeSp),
      x = "",
      y = "Occupancy"
    ) +
    theme_minimal()

  print(pOccu)
  # ---------------------------------------------
  # backTransform(m1, type="psi")
  # confint(backTransform(m1, type="psi"))
  # backTransform(m1, type="ext")
  return(data = df_plot)
}
```

```{r}
      cellcode20 = all_cells,
      date_year = all_years,
      fill = list(n_occurrences = 0)
    ) %>%
    mutate(detection = ifelse(n_occurrences > 0, 1, 0)) %>%
    pivot_wider(
      id_cols = cellcode20,
      names_from = date_year,
      values_from = detection
    ) %>%
    column_to_rownames("cellcode20") %>%
    as.matrix()

  # Covariata Anno (vettore atomico per NIMBLE)
  anni_num <- as.numeric(colnames(y_matrix))
  year_std <- as.vector(scale(anni_num))

  # Standardizzazione (vettoriale per evitare attributi scale)
  effort_vec <- as.numeric(effort_matrix)
  effort_std_mat <- matrix(
    (effort_vec - mean(effort_vec)) / sd(effort_vec),
    nrow = nrow(y_matrix),
    ncol = ncol(y_matrix)
  )

  # Dati presenza e effort
  nim_data <- list(
    y = y_matrix,
    effort = effort_std_mat,
    year = year_std
  )

  nim_constants <- list(
    n_sites = nrow(y_matrix),
    n_years = ncol(y_matrix)
  )

  # 2. DEFINIZIONE MODELLO
  code <- nimbleCode({
    # PRIORS
    alpha_psi ~ dnorm(0, sd = 2)
    beta_year_psi ~ dnorm(0, sd = 2)
    alpha_p ~ dnorm(0, sd = 2)
    beta_effort ~ dnorm(0, sd = 2)
    beta_year_p ~ dnorm(0, sd = 2)

    # LIKELIHOOD
    for (i in 1:n_sites) {
      for (t in 1:n_years) {
        # Occupancy
        logit(psi[i, t]) <- alpha_psi + beta_year_psi * year[t]
        z[i, t] ~ dbern(psi[i, t])

        # Detection
        logit(p[i, t]) <- alpha_p +
          beta_effort * effort[i, t] +
          beta_year_p * year[t]
        mu[i, t] <- z[i, t] * p[i, t]
        y[i, t] ~ dbern(mu[i, t])
      }
    }

    # DERIVED
    for (t in 1:n_years) {
      mean_psi[t] <- ilogit(alpha_psi + beta_year_psi * year[t])
    }
    mean_p <- ilogit(alpha_p)
  })

  # 3. INIZIALIZZAZIONE
  # Importante: z deve essere 1 se y è 1
  z_init <- y_matrix
  z_init[is.na(z_init)] <- 0

  inits <- list(
    alpha_psi = 0,
    beta_year_psi = 0,
    alpha_p = 0,
    beta_effort = 0,
    beta_year_p = 0,
    z = z_init
  )

  # 4. ESECUZIONE
  model <- nimbleModel(
    code,
    constants = nim_constants,
    data = nim_data,
    inits = inits
  )
  c_model <- compileNimble(model)

  mcmc_conf <- configureMCMC(
    model,
    monitors = c(
      'alpha_psi',
      'beta_year_psi',
      'alpha_p',
      'beta_effort',
      'mean_psi'
    )
  )
  mcmc <- buildMCMC(mcmc_conf)
  c_mcmc <- compileNimble(mcmc, project = model)

  samples <- runMCMC(
    c_mcmc,
    niter = 10000,
    nburnin = 2000,
    nchains = 3,
    samplesAsCodaMCMC = TRUE
  )

  # 5. RUN MCMC
  set.seed(123)
  samples <- runMCMC(
    c_mcmc,
    niter = 20000, # corretto da 10000
    nburnin = 5000,
    nchains = 3,
    thin = 10, # da 5
    samplesAsCodaMCMC = TRUE
  )

  # 6. CALCOLO DIAGNOSTICHE (R-hat)
  rhat_values <- gelman.diag(samples, multivariate = FALSE)
  rhat_df <- as.data.frame(rhat_values$psrf)

  # 7. RISULTATI E SALVATAGGIO
  # Creiamo una lista con tutto ciò che è importante
  risultati_finali <- list(
    specie = nomeSp,
    summary = summary(samples),
    rhat = rhat_df,
    samples = samples # salva i campioni completi
  )

  # Crea la cartella di output se non esiste
  if (!dir.exists("output_bayes")) {
    dir.create("output_bayes")
  }

  # Salva in formato RDS (il nome del file include il nome della specie)
  file_rds <- paste0("output_bayes/risultati_", nomeSp, ".rds")
  saveRDS(risultati_finali, file = file_rds)

  message(paste("Statistiche salvate in:", file_rds))

  # 8. PLOT TREND
  anni_num <- as.numeric(colnames(y_matrix))
  mean_psi_cols <- grep("mean_psi", colnames(as.matrix(samples)))
  mean_psi_samples <- as.matrix(samples)[, mean_psi_cols]

  est <- colMeans(mean_psi_samples)
  ci <- apply(mean_psi_samples, 2, quantile, probs = c(0.025, 0.975))

  # Generazione del grafico
  plot(
    anni_num,
    est,
    type = "l",
    pch = 16,
    ylim = c(0, 1),
    xlab = "Year",
    ylab = "Occupancy",
    main = paste("Trend Bayesiano:", nomeSp)
  )
  lines(anni_num, ci[1, ], lty = 2, col = "blue")
  lines(anni_num, ci[2, ], lty = 2, col = "blue")
  polygon(
    c(anni_num, rev(anni_num)),
    c(ci[1, ], rev(ci[2, ])),
    col = rgb(0, 0, 1, 0.05),
    border = NA
  )
}
```
