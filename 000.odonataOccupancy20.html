Sampling effort

Per tenere conto anche del differente sforzo di copertura delle celle del territorio nel corso degli anni, il modello lm è stato modificato in glm con una correzione derivante dall’utilizzo di un offset pari al log(n_sites_censiti) per ciascun anno. Il modello normalizza i conteggi per lo sforzo, così che la pendenza stimata per date_year rifletta la vera tendenza di espansione o contrazione e non solo la variazione derivante dal differente sforzo di campionamento.

Code 
```{r glm_offset}
# n_sites occupati da ciascuna specie per anno
species_sites <- dfo |>
  group_by(species, date_year) |>
  summarise(n_sites = n_distinct(cellcode20), .groups = "drop")

# n_sites monitorati per anno totali
effort_df <- dfo |>
  group_by(date_year) |>
  distinct(cellcode20) |>
  count(date_year, name = 'n_sites_censiti')

species_sites <- species_sites |>
  left_join(effort_df, by = "date_year")

results <- species_sites |>
  group_by(species) |>
  do({
    mod <- glm(
      n_sites ~ date_year,
      family = quasipoisson,
      offset = log(n_sites_censiti),
      data = .
    )
    tidy(mod) # restituisce coefficienti, SE, p-value
  }) |>
  ungroup()

species_trends <- results |>
  filter(term == "date_year") |>
  mutate(
    trend = case_when(
      estimate > 0 ~ "increase",
      estimate < 0 ~ "decrease",
      TRUE ~ "stable"
    )
  )
```

Uso di TRIM

Il dato utilizzato per la modellistica è il numero di occorrenze per cella per ciascun anno. Utilizzato il Model 2 con Trend lineare.

```{r TRIM}
nomeSp <- listaSpecie[5]

odoTrim <- function(nomeSp) {
  # celle per specie (by year)
  df_occu <- dfo |>
    filter(species == nomeSp) |>
    group_by(date_year, cellcode20, species) |>
    count(name = 'count') #occorrenze

  # celle visitate area di studio (by year)
  df_occuTot <- dfo |>
    as_tibble() |>
    count(date_year, cellcode20, name = 'n_segnalazioni') |>
    mutate(effort = 1 / (n_segnalazioni))

  # stima presenza/pseudo-assenza e celle non visitate
  df_occu <- df_occu |>
    #select(-nTot_occurrences) |>
    left_join(df_occuTot, by = c("cellcode20", "date_year")) |>
    mutate(count = replace_na(count, 0))
  # complete(cellcode20,date_year,
  #            fill = list(presence=NA))

  # Check: at least 3 years
  if (n_distinct(df_occu$date_year) < 3) {
    message("Not enough years for ", nomeSp, " — skipped")
    return(NULL)
  }
  # Check: at least 3 sites
  if (n_distinct(df_occu$cellcode20) < 3) {
    message("Not enough sites for ", nomeSp, " — skipped")
    return(NULL)
  }
  # Try to fit TRIM
  res <- tryCatch(
    {
      # effort / bilanciate rispetto al numero di celle visitate per l'anno
      m1 <- trim(
        count ~ cellcode20 + date_year,
        data = df_occu,
        stepwise = T,
        model = 2,
        changepoints = "all",
        serialcor = F,
        overdisp = TRUE
      )

      m2 <- trim(
        count ~ cellcode20 + date_year,
        data = df_occu,
        stepwise = T,
        weights = 'effort',
        model = 2,
        changepoints = "all",
        serialcor = F,
        overdisp = TRUE
      )
      summary(m1)
      out <- overall(m2)

      out$slope$species = nomeSp
      # Filename
      filename <- paste0("trim/trim_", gsub(" ", "_", nomeSp), ".pdf")
      # Save PDF
      #pdf(filename, width = 7, height = 5)
      layout(matrix(c(1, 2), 1, 2))
      plot(overall(m1), main = paste0(overall(m1)$slope$meaning))
      plot(overall(m2), main = paste0("w: ", overall(m2)$slope$meaning))
      title(nomeSp, line = -1, outer = TRUE)

      #plot(overall(m3), main = paste0(nomeSp,' - ',overall(m3)$slope$meaning))
      #dev.off()
      #message("Completed: ", nomeSp)
      return(out)
    },
    error = function(e) {
      message("Error for ", nomeSp, ": ", e$message)
      return(NULL)
    }
  )
  return(res)
}
```

```{r}
library(nimble)
# https://www.youtube.com/watch?v=kmMZpG4jK4k&t=192s
library(coda)

# source('R-Script/baseDati.R')
nomeSp = 'Trithemis annulata'

odoBayes <- function(nomeSp) {
  # 1. PREPARAZIONE DATI
  all_cells <- sort(unique(dfo$cellcode20))
  all_years <- sort(unique(dfo$date_year))

  # Matrice di rilevamento (y)
  y_matrix <- dfo %>%
    filter(species == nomeSp) %>%
    count(cellcode20, date_year, name = 'n_occurrences') %>%
    complete(
      cellcode20 = all_cells,
      date_year = all_years,
      fill = list(n_occurrences = 0)
    ) %>% # includi anche le celle mai occupate dalla specie
    mutate(detection = ifelse(n_occurrences > 0, 1, 0)) %>%
    pivot_wider(
      id_cols = cellcode20,
      names_from = date_year,
      values_from = detection
    ) %>%
    column_to_rownames("cellcode20") %>%
    as.matrix()

  # Matrice dello sforzo
  effort_matrix <- dfo %>%
    count(cellcode20, date_year, name = 'tot_occurrences') %>%
    complete(
      cellcode20 = all_cells,
      date_year = all_years,
      fill = list(tot_occurrences = 0)
    ) %>%
    pivot_wider(
      id_cols = cellcode20,
      names_from = date_year,
      values_from = tot_occurrences
    ) %>%
    column_to_rownames("cellcode20") %>%
    as.matrix()

  # Standardizzazione corretta (vettoriale per evitare attributi scale)
  effort_vec <- as.numeric(effort_matrix)
  effort_std_mat <- matrix(
    (effort_vec - mean(effort_vec)) / sd(effort_vec),
    nrow = nrow(y_matrix),
    ncol = ncol(y_matrix)
  )

  # Covariata Anno (vettore atomico per NIMBLE)
  anni_num <- as.numeric(colnames(y_matrix))
  year_std <- as.vector(scale(anni_num))

  # Dati e Costanti
  nim_data <- list(
    y = y_matrix,
    effort = effort_std_mat,
    year = year_std
  )

  nim_constants <- list(
    n_sites = nrow(y_matrix),
    n_years = ncol(y_matrix)
  )

  # 2. DEFINIZIONE MODELLO
  code <- nimbleCode({
    # PRIORS
    alpha_psi ~ dnorm(0, sd = 2)
    beta_year_psi ~ dnorm(0, sd = 2)
    alpha_p ~ dnorm(0, sd = 2)
    beta_effort ~ dnorm(0, sd = 2)
    beta_year_p ~ dnorm(0, sd = 2)

    # LIKELIHOOD
    for (i in 1:n_sites) {
      for (t in 1:n_years) {
        # Occupancy
        logit(psi[i, t]) <- alpha_psi + beta_year_psi * year[t]
        z[i, t] ~ dbern(psi[i, t])

        # Detection
        logit(p[i, t]) <- alpha_p +
          beta_effort * effort[i, t] +
          beta_year_p * year[t]
        mu[i, t] <- z[i, t] * p[i, t]
        y[i, t] ~ dbern(mu[i, t])
      }
    }

    # DERIVED
    for (t in 1:n_years) {
      mean_psi[t] <- ilogit(alpha_psi + beta_year_psi * year[t])
    }
    mean_p <- ilogit(alpha_p)
  })

  # 3. INIZIALIZZAZIONE
  # Importante: z deve essere 1 se y è 1
  z_init <- y_matrix
  z_init[is.na(z_init)] <- 0

  inits <- list(
    alpha_psi = 0,
    beta_year_psi = 0,
    alpha_p = 0,
    beta_effort = 0,
    beta_year_p = 0,
    z = z_init
  )

  # 4. ESECUZIONE
  model <- nimbleModel(
    code,
    constants = nim_constants,
    data = nim_data,
    inits = inits
  )
  c_model <- compileNimble(model)

  mcmc_conf <- configureMCMC(
    model,
    monitors = c(
      'alpha_psi',
      'beta_year_psi',
      'alpha_p',
      'beta_effort',
      'mean_psi'
    )
  )
  mcmc <- buildMCMC(mcmc_conf)
  c_mcmc <- compileNimble(mcmc, project = model)

  samples <- runMCMC(
    c_mcmc,
    niter = 10000,
    nburnin = 2000,
    nchains = 3,
    samplesAsCodaMCMC = TRUE
  )

  # 5. RUN MCMC
  set.seed(123)
  samples <- runMCMC(
    c_mcmc,
    niter = 10000,
    nburnin = 2000,
    nchains = 3,
    thin = 5,
    samplesAsCodaMCMC = TRUE
  )

  # 6. CALCOLO DIAGNOSTICHE (R-hat)
  rhat_values <- gelman.diag(samples, multivariate = FALSE)
  rhat_df <- as.data.frame(rhat_values$psrf)

  # 7. RISULTATI E SALVATAGGIO
  # Creiamo una lista con tutto ciò che è importante
  risultati_finali <- list(
    specie = nomeSp,
    summary = summary(samples),
    rhat = rhat_df,
    samples = samples # opzionale: salva i campioni completi
  )

  # Crea la cartella di output se non esiste
  if (!dir.exists("output_bayes")) {
    dir.create("output_bayes")
  }

  # Salva in formato RDS (il nome del file include il nome della specie)
  file_rds <- paste0("output_bayes/risultati_", nomeSp, ".rds")
  saveRDS(risultati_finali, file = file_rds)

  message(paste("Statistiche salvate in:", file_rds))

  # 8. PLOT TREND (Output visivo)
  anni_num <- as.numeric(colnames(y_matrix))
  mean_psi_cols <- grep("mean_psi", colnames(as.matrix(samples)))
  mean_psi_samples <- as.matrix(samples)[, mean_psi_cols]

  est <- colMeans(mean_psi_samples)
  ci <- apply(mean_psi_samples, 2, quantile, probs = c(0.025, 0.975))

  # Generazione del grafico
  plot(
    anni_num,
    est,
    type = "l",
    pch = 16,
    ylim = c(0, 1),
    xlab = "Year",
    ylab = "Occupancy",
    main = paste("Trend Bayesiano:", nomeSp)
  )
  lines(anni_num, ci[1, ], lty = 2, col = "blue")
  lines(anni_num, ci[2, ], lty = 2, col = "blue")
  polygon(
    c(anni_num, rev(anni_num)),
    c(ci[1, ], rev(ci[2, ])),
    col = rgb(0, 0, 1, 0.05),
    border = NA
  )

  # La funzione restituisce il grafico a video e il file su disco
}
```
