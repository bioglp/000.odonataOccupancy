make_matrix <- function(df_year) {
  df_year |>
    group_by(cellcode, species) |>
    summarise(count = n(), .groups = "drop") |> # conteggio osservazioni
    pivot_wider(names_from = species, values_from = count, values_fill = 0) |>
    column_to_rownames("cellcode") |>
    as.matrix()
}

# Lista degli anni
dfo <- dfo |>
  mutate(
    year_group = paste0(
      floor(date_year / 5) * 5,
      "-",
      floor(date_year / 5) * 5 + 4
    )
  )
dfo$year_group[dfo$year_group == "2020-2024"] <- "2020-2025"
dfo$year_group[dfo$year_group == "2025-2029"] <- "2020-2025"


# Funzione per creare matrice sito × specie con conteggi
make_matrix <- function(df_group) {
  df_group |>
    group_by(cellcode, species) |>
    summarise(count = n(), .groups = "drop") |>

Sampling effort

Per tenere conto anche del differente sforzo di copertura delle celle del territorio nel corso degli anni, il modello lm è stato modificato in glm con una correzione derivante dall’utilizzo di un offset pari al log(tot_occurrences) per ciascun anno. Il modello normalizza i conteggi per lo sforzo, così che la pendenza stimata per date_year rifletta la vera tendenza di espansione o contrazione e non solo la variazione derivante dal differente sforzo di campionamento.

Code 
```{r glm_offset}
#| eval: true
#| include: true

# 1. PREPARAZIONE DATI
all_cells <- sort(unique(dfo$cellcode))
all_years <- sort(unique(dfo$date_year))

# Matrice dello sforzo
effort_df <- dfo %>%
  count(cellcode, date_year, name = 'tot_occurrences') %>%
  complete(
    cellcode = all_cells,
    date_year = all_years,
    fill = list(tot_occurrences = 0)
  )
# specie test
# nomeSp <- listaSpecie[1]

glmSpecies <- function(nomeSp) {
  y_df <- dfo %>%
    filter(species == nomeSp) %>%
    count(cellcode, date_year, name = 'n_occurrences') %>%
    complete(
      # includi anche le celle mai occupate dalla specie
      cellcode = all_cells,
      date_year = all_years,
      fill = list(n_occurrences = 0)
    ) %>%
    mutate(detection = ifelse(n_occurrences > 0, 1, 0))

  species_sites <- merge(y_df, effort_df)

  model_offset <- glm(
    detection ~ date_year + offset(log(tot_occurrences + 1)),
    family = binomial,
    data = species_sites
  )

  sp_results <- tidy(model_offset)
  sp_results$species = nomeSp
  return(sp_results)
}

#glmSpecies(listaSpecie[5])
results <- map(listaSpecie, glmSpecies) |> list_rbind()

species_trends <- results |>
  filter(term == "date_year") |>
  mutate(
    trend = case_when(
      estimate > 0 ~ "increase",
      estimate < 0 ~ "decrease",
      TRUE ~ "stable"
    )
  )
```
